    private static void TestFixture(Fixture f)
    {
        long IdentityMap(long key) => key;
        Check(f.Run(f.Build(IdentityMap), null), f.Name + " default legacy bytes accepted with payload fields intact");
        var calls = new List<long>();
        long TraceIdentity(long key) { calls.Add(key); return key; }
        Check(f.Run(f.Build(IdentityMap), TraceIdentity) && calls.SequenceEqual(f.Keys), f.Name + " explicit identity matches default and traces exact keys");
        var legacy = f.Build(IdentityMap); var candidate = f.Build(Map);
        var normalized = candidate.Select(b => (byte[])b.Clone()).ToArray();
        foreach (int buffer in f.GuardBuffers) Array.Copy(legacy[buffer], 0, normalized[buffer], 0, 8);
        Check(normalized.Zip(legacy).All(pair => pair.First.SequenceEqual(pair.Second)), f.Name + " candidate changes only vtable bytes");
        Check(!f.Run(candidate, null), f.Name + " default legacy guard rejects candidate bytes");
        calls.Clear();
        long TraceCandidate(long key) { calls.Add(key); return Map(key); }
        Check(f.Run(candidate, TraceCandidate) && calls.SequenceEqual(f.Keys), f.Name + " mapped candidate accepted and exact dependency keys invoked");
        for (int i = 0; i < f.Keys.Length; i++)
        {
            var wrong = f.Build(Map); Q(wrong[f.GuardBuffers[i]], 0, Image + 0x7FFF0000);
            Check(!f.Run(wrong, Map), f.Name + " wrong type independently rejected for key " + f.Keys[i]);
            var mixed = f.Build(Map); Q(mixed[f.GuardBuffers[i]], 0, Image + f.Keys[i]);
            Check(!f.Run(mixed, Map), f.Name + " legacy type cannot pass mapped guard for key " + f.Keys[i]);
            long failKey = f.Keys[i]; var sentinel = new MapperFailure(); bool propagated = false;
            try { f.Run(f.Build(Map), key => key == failKey ? throw sentinel : Map(key)); }
            catch (MapperFailure ex) { propagated = ReferenceEquals(ex, sentinel); }
            Check(propagated, f.Name + " mapper exception propagates unchanged for key " + failKey);
        }
        var a = new MappingContext(0); var b = new MappingContext(0x2000);
        Check(f.Run(f.Build(a.Resolve), a.Resolve) && f.Run(f.Build(b.Resolve), b.Resolve) && f.Run(f.Build(a.Resolve), a.Resolve), f.Name + " interleaved mapping contexts remain isolated");
        Check(!f.Run(f.Build(a.Resolve), b.Resolve), f.Name + " different mapping context cannot accept other context bytes");
        int invoked = 0;
        long Unexpected(long key) { invoked++; throw new MapperFailure(); }
        var shortBytes = f.Build(Map); shortBytes[0] = new byte[f.MinimumFirstLength - 1];
        Check(!f.Run(shortBytes, Unexpected) && invoked == 0, f.Name + " short first buffer exits before mapper");
        var nullBytes = f.Build(Map); nullBytes[0] = null; bool nullExpected;
        if (f.NullThrows)
        {
            nullExpected = false;
            try { f.Run(nullBytes, Unexpected); } catch (NullReferenceException) { nullExpected = true; }
        }
        else nullExpected = !f.Run(nullBytes, Unexpected);
        Check(nullExpected && invoked == 0, f.Name + " existing null behavior preserved without mapper invocation");
        var noEvent = f.Build(Map);
        if (f.Name == "SupportActive") U(noEvent[0], 68, uint.MaxValue);
        else if (f.Name == "AirCast") noEvent[0][76] = 2;
        else U(noEvent[0], 160, 0x4000);
        Check(!f.Run(noEvent, Map), f.Name + " correctly mapped no-event negative control remains false/null");
    }

    private static void TestCapturedProjectile(string path)
    {
        byte[] file = File.ReadAllBytes(path);
        using var document = JsonDocument.Parse(file);
        var record = document.RootElement.GetProperty("records")[0];
        var body = Convert.FromHexString(record.GetProperty("rawBodyHex").GetString());
        var actor = Convert.FromHexString(record.GetProperty("rawActorHex").GetString());
        var ext = Convert.FromHexString(record.GetProperty("rawExtHex").GetString());
        long entity = Convert.ToInt64(record.GetProperty("entity").GetString().Substring(2), 16);
        uint id = record.GetProperty("generation").GetUInt32(), index = record.GetProperty("registryIndex").GetUInt32();
        long arena = entity - BitConverter.ToUInt32(actor, 36);
        long image = BitConverter.ToInt64(actor, 0) - Map(65313128);
        Check(body.Length == 392 && actor.Length == 80 && ext.Length == 208, "captured projectile retains exact 392/80/208 raw lengths");
        Check(arena == entity - BitConverter.ToUInt32(ext, 8) && arena == 0x17C00000000, "captured actor/ext backrefs derive same supplied arena context");
        Check(image == BitConverter.ToInt64(ext, 0) - Map(65307976) && image == 0x7FF7A37E0000, "captured actor/ext vtables derive same supplied image context");
        Check(ProjectileLogic.Mortar(record.GetProperty("ext").GetProperty("weapon").GetString()), "captured weapon selects actual mortar predicate");
        Check(!ProjectileLogic.Decode(body, actor, ext, image, arena, entity, id, index, out _, mortar: true), "captured candidate bytes rejected by default legacy guards");
        var calls = new List<long>();
        long Trace(long key) { calls.Add(key); return Map(key); }
        bool decoded = ProjectileLogic.Decode(body, actor, ext, image, arena, entity, id, index, out var impact, mortar: true, resolveRva: Trace);
        Check(decoded && calls.SequenceEqual(new long[] {65313128, 65307976}), "captured actual raw bytes accepted by mapped mortar decoder with exact keys");
        Check(impact != null && impact.Entity == entity && impact.Id == id && impact.Index == index && impact.SourceKey == ((1000008229UL << 32) | 3061) && impact.Radius == 4.5f && impact.X == record.GetProperty("actor").GetProperty("position20")[0].GetSingle() && impact.Y == record.GetProperty("actor").GetProperty("position20")[1].GetSingle() && impact.Z == record.GetProperty("actor").GetProperty("position20")[2].GetSingle(), "captured decoder payload matches retained event metadata");
        Check(!ProjectileLogic.Decode(body, actor, ext, image, arena, entity, id, index, out _, resolveRva: Map), "captured state64=3 requires explicit mortar or infantry context");
